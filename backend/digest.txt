# Місто Сили — Інформаційна вебсистема

Вебсайт громадської організації «Місто Сили» з адміністративними та соціальними сервісами.

## Стек технологій

| Компонент | Технологія |
|-----------|------------|
| Бекенд | Python 3.12, Django 5, DRF |
| Фронтенд | Next.js 15, TypeScript |
| БД | PostgreSQL |
| CMS | Django Admin + CKEditor 5 |
| Хостинг | cPanel |
| CDN / Security | Cloudflare |

## Структура проекту

```
diploma/
  backend/          ← Django + DRF
    accounts/       ← користувачі, авторизація
    core/           ← утиліти, middleware
    pages/          ← конструктор сторінок
    reports/        ← звіти
    schedule/       ← події
    seo/            ← sitemap, robots, redirects
    scripts/        ← backup.sh
  frontend/         ← Next.js
    src/
      app/          ← сторінки (App Router)
      components/   ← компоненти
      lib/          ← утиліти
```


# Інструкція з деплою

## Вимоги до сервера

- cPanel хостинг з підтримкою Python 3.12+ і Node.js 20+
- PostgreSQL (через cPanel або окремий сервер)
- Cloudflare (DNS + WAF + Access)

## Підготовка Cloudflare

1. Перенести DNS домену на Cloudflare
2. Увімкнути проксі (помаранчева хмара) для `app.domain` і `api.domain`
3. SSL/TLS → Full (strict)
4. Edge Certificates → Always Use HTTPS → ON
5. Створити Cloudflare Access Application:
   - Path: `api.domain/admin/*`
   - Policy: email domain = `mistosyly.org` + TOTP

## Деплой бекенду (cPanel)

### 1. Python App

- cPanel → Setup Python App
- Python version: 3.12
- Application root: `backend/`
- Application URL: `api.domain`
- Application startup file: `passenger_wsgi.py`

### 2. Файли

```bash
git clone <repo> ~/backend
cd ~/backend
pip install -r requirements.txt
```

### 3. Змінні середовища

В cPanel → Python App → Environment variables додати всі з `.env`.  
Або створити `.env` файл на сервері (не в репозиторії).

### 4. Міграції і початкові дані

```bash
python manage.py migrate
python manage.py createsuperuser
python manage.py seed_pages
python manage.py collectstatic --no-input
```

### 5. passenger_wsgi.py

```python
import os

os.environ['DJANGO_SETTINGS_MODULE'] = 'backend.settings.prod'
```

### 1. Node.js App

- cPanel → Setup Node.js App
- Node.js version: 20
- Application root: `frontend/`
- Application URL: `app.domain`
- Application startup file: `server.js`

### 2. Збірка

### 3. server.js

```javascript
const { createServer } = require('http');
const { parse } = require('url');
const next = require('next');

const app = next({ dev: false });
const handle = app.getRequestHandler();

app.prepare().then(() => {
  createServer((req, res) => {
    const parsedUrl = parse(req.url, true);
    handle(req, res, parsedUrl);
  }).listen(process.env.PORT || 3000);
});
```

from django.contrib.auth.models import AbstractBaseUser, BaseUserManager, PermissionsMixin
from django.db import models


class UserManager(BaseUserManager):
    def create_user(self, email, password=None, **extra_fields):
        if not email:
            raise ValueError("Email обов'язковий")
        email = self.normalize_email(email)
        user = self.model(email=email, **extra_fields)
        user.set_password(password)
        user.save(using=self._db)
        return user

    def create_superuser(self, email, password=None, **extra_fields):
        extra_fields.setdefault("role", "root")
        extra_fields.setdefault("is_staff", True)
        extra_fields.setdefault("is_superuser", True)
        return self.create_user(email, password, **extra_fields)


class User(AbstractBaseUser, PermissionsMixin):
    class Role(models.TextChoices):
        ROOT = "root",   "Root (повний доступ)"
        EDITOR = "editor", "Редактор"

    EDITOR_APPS = {"reports", "schedule", "pages"}

    email = models.EmailField(unique=True, verbose_name="Email")
    name = models.CharField(max_length=100, verbose_name="Ім'я")
    role = models.CharField(
        max_length=10,
        choices=Role.choices,
        default=Role.EDITOR,
        verbose_name="Роль",
    )

    google_sub = models.CharField(
        max_length=200, blank=True, null=True, unique=True,
        verbose_name="Google UID (sub)",
        help_text="Заповнюється автоматично при першому вході через Google.",
    )
    google_login_allowed = models.BooleanField(
        default=False,
        verbose_name="Дозволити вхід через Google",
        help_text="Root вручну дозволяє конкретному юзеру логінитись через Google.",
    )

    is_active = models.BooleanField(default=True,  verbose_name="Активний")
    is_staff = models.BooleanField(default=False, verbose_name="Доступ до адмінки Django")

    date_joined = models.DateTimeField(auto_now_add=True, verbose_name="Дата реєстрації")

    USERNAME_FIELD = "email"
    REQUIRED_FIELDS = ["name"]

    objects = UserManager()

    class Meta:
        verbose_name = "Користувач"
        verbose_name_plural = "Користувачі"

    def __str__(self):
        return f"{self.name} <{self.email}> [{self.get_role_display()}]"

    @property
    def is_root(self):
        return self.role == self.Role.ROOT

    @property
    def is_editor(self):
        return self.role in (self.Role.ROOT, self.Role.EDITOR)

    def has_perm(self, perm, obj=None):
        if not self.is_active:
            return False
        if self.is_superuser or self.is_root:
            return True
        if self.is_editor:
            app_label = perm.split(".")[0] if "." in perm else perm
            return app_label in self.EDITOR_APPS
        return False

    def has_module_perms(self, app_label):
        if not self.is_active:
            return False
        if self.is_superuser or self.is_root:
            return True
        if self.is_editor:
            return app_label in self.EDITOR_APPS
        return False

    def save(self, *args, **kwargs):
        if self.role in (self.Role.ROOT, self.Role.EDITOR):
            self.is_staff = True
        if self.role == self.Role.ROOT:
            self.is_superuser = True
        super().save(*args, **kwargs)

from social_core.exceptions import AuthForbidden
ALLOWED_DOMAINS = ['gmail.com', 'mistosyly.org']

def require_allowlist(backend, user, response, social, *args, **kwargs):
    if not response.get('email_verified'):
        raise AuthForbidden(backend)
    
    email = response.get('email', '')
    domain = email.split('@')[-1] if '@' in email else ''
    if domain not in ALLOWED_DOMAINS:
        raise AuthForbidden(backend)

    if not user:
        raise AuthForbidden(backend)

    if not user.google_login_allowed:
        raise AuthForbidden(backend)

    google_sub = response.get('sub')
    if google_sub and not user.google_sub:
        user.google_sub = google_sub
        user.save(update_fields=['google_sub'])

    return {'user': user}

def set_staff_flag(backend, user, *args, **kwargs):
    if user and not user.is_staff:
        user.is_staff = True
        user.save(update_fields=['is_staff'])

            profile_form = ProfileForm(request.POST, instance=request.user)
            if profile_form.is_valid():
                profile_form.save()
                messages.success(request, "Профіль оновлено.")
                return redirect("accounts:profile")

        elif "change_password" in request.POST:
            password_form = ChangePasswordForm(user=request.user, data=request.POST)
            if password_form.is_valid():
                password_form.save()
                messages.success(request, "Пароль змінено. Увійдіть знову.")
                return redirect("admin:login")

    totp_device = TOTPDevice.objects.filter(user=request.user, confirmed=True).first()

    return render(request, "admin/profile.html", admin_template_context(
        request,
        title="Мій профіль",
        profile_form=profile_form,
        password_form=password_form,
        totp_enabled=totp_device is not None,
        has_permission=True,
    ))


def _render_2fa_setup(request, device, extra=None):
    img = qrcode.make(device.config_url)
    buf = io.BytesIO()
    img.save(buf, format="PNG")
    qr_b64 = base64.b64encode(buf.getvalue()).decode("utf-8")
    ctx = admin_template_context(
        request,
        title="Налаштування 2FA",
        qr_b64=qr_b64,
        secret_key=device.bin_key.hex(),
        has_permission=True,
    )
    if extra:
        ctx.update(extra)
    return render(request, "admin/setup_2fa.html", ctx)


@staff_member_required
def setup_2fa_view(request):
    if request.method == "POST":
        code   = request.POST.get("code", "").strip()
        device = TOTPDevice.objects.filter(user=request.user, confirmed=False).first()

        if not device:
            device = TOTPDevice.objects.create(
                user=request.user,
                name=f"{request.user.email} TOTP",
                confirmed=False,
            )
            return _render_2fa_setup(request, device)

        if device.verify_token(code):
            device.confirmed = True
            device.save()
            messages.success(request, "2FA успішно налаштовано!")
            return redirect("accounts:profile")

        return _render_2fa_setup(request, device, {"error": "Невірний код. Спробуйте ще раз."})

    TOTPDevice.objects.filter(user=request.user, confirmed=False).delete()
    device = TOTPDevice.objects.create(
        user=request.user,
        name=f"{request.user.email} TOTP",
        confirmed=False,
    )
    return _render_2fa_setup(request, device)


@staff_member_required
def disable_2fa_view(request):
    if request.method == "POST":
        TOTPDevice.objects.filter(user=request.user).delete()
        messages.success(request, "2FA вимкнено.")
    return redirect("accounts:profile")


def verify_2fa_view(request):
    if not request.user.is_authenticated:
        return redirect("admin:login")

    if request.user.is_verified():
        return redirect("admin:index")

    has_device = TOTPDevice.objects.filter(
        user=request.user, confirmed=True
    ).exists()
    if not has_device:
        return redirect("admin:index")

    error = None
    if request.method == "POST":
        code = request.POST.get("code", "").strip()
        device = TOTPDevice.objects.filter(
            user=request.user, confirmed=True
        ).first()

        if device and device.verify_token(code):
            device.persistent_id
            request.user.otp_device = device
            from django_otp import login as otp_login
            otp_login(request, device)
            next_url = request.GET.get("next", "/admin/")
            return redirect(next_url)

        error = "Невірний код. Спробуйте ще раз."

    return render(request, "admin/verify_2fa.html", admin_template_context(
        request,
        title="Підтвердження входу",
        error=error,
        has_permission=False,
    ))

class ShortTitleMixin:
    """Короткі заголовки вкладок замість «Виберіть X щоб змінити»."""

    admin_list_title: str | None = None
    admin_add_title: str | None = None

    def _list_title(self) -> str:
        if self.admin_list_title:
            return self.admin_list_title
        opts = self.model._meta
        return str(opts.verbose_name_plural or opts.verbose_name)

    def _add_title(self) -> str:
        if self.admin_add_title:
            return self.admin_add_title
        return f'Додати {self.model._meta.verbose_name}'

    def changelist_view(self, request, extra_context=None):
        extra_context = extra_context or {}
        extra_context.setdefault('title', self._list_title())
        return super().changelist_view(request, extra_context=extra_context)

    def add_view(self, request, form_url='', extra_context=None):
        extra_context = extra_context or {}
        extra_context.setdefault('title', self._add_title())
        return super().add_view(request, form_url, extra_context=extra_context)

    def change_view(self, request, object_id, form_url='', extra_context=None):
        extra_context = extra_context or {}
        obj = self.get_object(request, object_id)
        if obj is not None:
            extra_context.setdefault('title', str(obj))
        return super().change_view(request, object_id, form_url, extra_context=extra_context)

import time

from django.shortcuts import redirect
from django_otp.plugins.otp_totp.models import TOTPDevice
from django.contrib.auth import logout


class Require2FAMiddleware:
    EXEMPT_URLS = (
        '/admin/login/',
        '/admin/logout/',
        '/admin/profile/2fa/verify/',
        '/auth/',
    )

    def __init__(self, get_response):
        self.get_response = get_response

    def __call__(self, request):
        if (
            request.path.startswith('/admin/')
            and not any(request.path.startswith(u) for u in self.EXEMPT_URLS)
            and hasattr(request, 'user')
            and request.user.is_authenticated
            and request.user.is_staff
        ):

            has_device = TOTPDevice.objects.filter(
                user=request.user, confirmed=True
            ).exists()

            if has_device and not request.user.is_verified():
                return redirect(
                    f'/admin/profile/2fa/verify/?next={request.path}'
                )

        return self.get_response(request)

class SessionTimeoutMiddleware:
    def __init__(self, get_response):
        self.get_response = get_response
    
    def __call__(self, request):
        if request.user.is_authenticated:
            last_activity = request.session.get('last_activity')
            now = time.time()

            if last_activity and (now - last_activity > 60 * 60 * 8):
                logout(request)
                return redirect('admin/login/?timeout=1')
            
            request.session['last_activity'] = now
        
        return self.get_response(request)

class SearchView(APIView):
    throttle_classes = [SearchRateThrottle]

    def get(self, request):
        query = request.GET.get("q", "").strip()
        filter_type = request.GET.get("type", "")

        if len(query) < 2:
            return Response({
                "query": query,
                "results": [],
                "total": 0,
                "error": "Запит має містити мінімум 2 символи.",
            })

        results = []

        if not filter_type or filter_type == "report":
            reports = Report.objects.filter(
                status=Report.Status.PUBLISHED
            ).filter(
                Q(title__icontains=query) |
                Q(preview__icontains=query) |
                Q(content__icontains=query) |
                Q(slug__icontains=query)
            ).distinct()[:10]
            results += ReportSearchSerializer(
                reports, many=True, context={"request": request}
            ).data

        if not filter_type or filter_type == "event":
            events = Event.objects.filter(
                status=Event.Status.PUBLISHED
            ).filter(
                Q(title__icontains=query) |
                Q(description__icontains=query) |
                Q(location__icontains=query) |
                Q(slug__icontains=query)
            ).distinct()[:10]
            results += EventSearchSerializer(
                events, many=True, context={"request": request}
            ).data

        if not filter_type or filter_type == "page":
            pages = Page.objects.filter(
                is_active=True
            ).filter(
                Q(title__icontains=query) |
                Q(meta_title__icontains=query) |
                Q(meta_description__icontains=query) |
                Q(sections__title__icontains=query) |
                Q(sections__subtitle__icontains=query) |
                Q(sections__content__icontains=query)
            ).distinct()[:5]
            results += PageSearchSerializer(
                pages, many=True, context={"request": request}
            ).data

        return Response({
            "query": query,
            "results": results,
            "total": len(results),
        })

import os

def safe_filename(filename: str) -> str:
    name, ext = os.path.splitext(filename)
    safe = slugify(unidecode(name)) or f"file_{uuid.uuid4().hex[:8]}"
    return f"{safe}_{uuid.uuid4().hex[:8]}{ext.lower()}"


def _upload_to_generic(folder: str, instance, filename: str) -> str:
    folder = folder.strip('/')
    return posixpath.join(folder, safe_filename(filename))


def upload_to_factory(folder: str):
    def upload_to(instance, filename):
        return _upload_to_generic(folder, instance, filename)

    upload_to.__name__ = f'upload_to_{slugify(folder.replace("/", "_"))}'
    upload_to.deconstruct = lambda: (
        f'core.utils.upload_to_factory("{folder}")',
        [],
        {}
    )
    return upload_to

def slugify(text):
    text = unidecode(text)
    text = text.lower()
    text = re.sub(r'[^a-z0-9\s-]', '', text)
    text = re.sub(r'\s+', '-', text)

def dates_validator(date1, date2):
    try:
        date_1 = datetime.strptime(str(date1), "%Y-%m-%d").date() if date1 else None
        date_2 = datetime.strptime(str(date2), "%Y-%m-%d").date() if date2 else None
    except (ValueError, TypeError):
        return False

    if date_1 is None or date_2 is None:
        return True

    return date_1 <= date_2

from rest_framework import generics

from rest_framework.views import APIView
from django.shortcuts import get_object_or_404

from .models import Page, ContactInfo
from .serializers import PageSerializer, ContactInfoSerializer, PageNavSerializer


class PageDetailView(generics.RetrieveAPIView):
    serializer_class = PageSerializer
    lookup_field = "key"

    def get_queryset(self):
        return Page.objects.prefetch_related("sections").filter(is_active=True)

    def get_object(self):
        queryset = self.get_queryset()
        key = self.kwargs.get("key")
        try:
            page = queryset.get(key=key)
        except Page.DoesNotExist:
            raise NotFound(f"Сторінку '{key}' не знайдено або вона неактивна.")

        page.get_visible_sections = lambda: page.sections.filter(is_visible=True).order_by("order")
        return page

from django.db import models
from django_ckeditor_5.fields import CKEditor5Field
from core.validators import validate_file_size
from core.utils import upload_to_factory, slugify
from django.utils import timezone

class Report(models.Model):
    class Status(models.TextChoices):
        DRAFT = "draft", "Чернетка"
        PUBLISHED = "published", "Опубліковано"
        HIDDEN = "hidden", "Скрито"

    title = models.CharField(max_length=200)
    slug = models.SlugField(max_length=200, unique=True)
    preview = models.TextField(max_length=200)
    content = CKEditor5Field('Content', config_name='extends')

    date_created = models.DateTimeField(auto_now_add=True)
    date_edited = models.DateTimeField(auto_now=True)
    date_published = models.DateTimeField(null=True, blank=True)

    status = models.CharField(
        max_length=10,
        db_index=True,
        choices=Status.choices,
        default=Status.DRAFT,
    )

    image = models.ImageField(
        upload_to=upload_to_factory('images/posts'),
        validators=[validate_file_size],
        blank=True,
    )

    class Meta:
        verbose_name = 'Звіт'
        verbose_name_plural = 'Звіти'
        ordering = ['-date_published']

    def save(self, *args, **kwargs):
        if not self.slug:
            base_slug = slugify(self.title) or "report"
            slug_candidate = base_slug
            counter = 2

            while Report.objects.filter(slug=slug_candidate).exclude(pk=self.pk).exists():
                slug_candidate = f"{base_slug}-{counter}"
                counter += 1

            self.slug = slug_candidate
        if self.status == self.Status.PUBLISHED and not self.date_published:
            self.date_published = timezone.now()
        super().save(*args, **kwargs)

# =============================================================
# Скрипт резервного копіювання PostgreSQL БД
# Розташування: backend/scripts/backup.sh
# Запуск вручну: bash scripts/backup.sh
# Автоматичний запуск: додати в cron (див. нижче)
# =============================================================

# Bash не підхоплює .env сам. Не використовуємо `source` для всього файлу:
# — CRLF (Windows) лишає \r у значеннях і ламає pg_dump;
# — рядки на кшталт SECRET_KEY=...&... інтерпретуються bash як команди.
load_backup_env() {
  local env_file="$1" line key value
  [ -f "$env_file" ] || return 0
  while IFS= read -r line || [ -n "$line" ]; do
    line="${line%$'\r'}"
    [[ "$line" == $'\ufeff'* ]] && line="${line#$'\ufeff'}"
    [[ -z "$line" || "$line" =~ ^[[:space:]]*# ]] && continue
    case "$line" in
      POSTGRES_DB=*|POSTGRES_USER=*|POSTGRES_PASSWORD=*|POSTGRES_HOST=*|POSTGRES_PORT=*|BACKUP_DIR=*|VERIFY_RESTORE_AFTER_BACKUP=*)
        key="${line%%=*}"
        value="${line#*=}"
        export "$key=$value"
        ;;
    esac
  done < "$env_file"
}

SCRIPT_DIR="$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd)"
load_backup_env "$SCRIPT_DIR/../.env"

set -euo pipefail

# --- Налаштування (береться з .env або задається тут) ---
DB_NAME="${POSTGRES_DB:-mydb}"
DB_USER="${POSTGRES_USER:-myuser}"
DB_HOST="${POSTGRES_HOST:-localhost}"
DB_PORT="${POSTGRES_PORT:-5432}"
# За замовчуванням — backend/backups/db (поруч із проєктом). Інакше задайте BACKUP_DIR у .env.
BACKUP_DIR="${BACKUP_DIR:-$SCRIPT_DIR/../backups/db}"
KEEP_DAYS=30  # скільки днів зберігати бекапи
# Перевірка відновлення: 1-го числа кожного місяця, або якщо в .env VERIFY_RESTORE_AFTER_BACKUP=1 (кожен запуск)

# --- Підготовка ---
mkdir -p "$BACKUP_DIR"
echo "[$(date)] Каталог бекапів: $BACKUP_DIR"
TIMESTAMP=$(date +"%Y-%m-%d_%H-%M-%S")
BACKUP_FILE="$BACKUP_DIR/${DB_NAME}_${TIMESTAMP}.sql.gz"

echo "[$(date)] Починаємо резервне копіювання БД: $DB_NAME"

# --- Створення бекапу ---
PGPASSWORD="$POSTGRES_PASSWORD" pg_dump \
    -h "$DB_HOST" \
    -p "$DB_PORT" \
    -U "$DB_USER" \
    -d "$DB_NAME" \
    --no-password \
    | gzip > "$BACKUP_FILE"

from django.http import HttpResponse, HttpResponsePermanentRedirect, HttpResponseRedirect, Http404
from django.views import View
from rest_framework.response import Response
from rest_framework.views import APIView
from .models import Redirect

class RobotsView(View):
    def get(self, request):
        lines = [
            "User-agent: *",
            "Allow: /",
            "Disallow: /admin/",
            "Disallow: /auth/",
            "Disallow: /api/",
            "",
            f"Sitemap: {request.scheme}://{request.get_host()}/sitemap.xml",
        ]
        return HttpResponse(
            "\n".join(lines), 
            content_type="text/plain"
        )

class RedirectMiddlewareView(View):
    def get(self, request, path):
        candidates = [
            f"/{path}",
            f"/{path}/",
            f"{path}",
        ]
        for from_path in candidates:
            try:
                redirect = Redirect.objects.get(from_path=from_path, is_active=True)
                if redirect.status_code == 301:
                    return HttpResponsePermanentRedirect(redirect.to_url)
                return HttpResponseRedirect(redirect.to_url)
            except Redirect.DoesNotExist:
                continue
        raise Http404


class RedirectListAPIView(APIView):
    """Список активних редиректів для Next.js proxy."""

    def get(self, request):
        rows = Redirect.objects.filter(is_active=True).values(
            "from_path", "to_url", "status_code"
        )
        return Response(list(rows))

'use strict';

const SECTION_FIELDS = {
    hero: ['field-title', 'field-subtitle', 'field-image', 'field-image_alt', 'field-button_text', 'field-button_url'],
    text: ['field-title', 'field-subtitle', 'field-content', 'field-image', 'field-image_alt'],
    gallery: ['field-title', 'field-subtitle'],
    contacts: ['field-title', 'field-content'],
    cta: ['field-title', 'field-subtitle', 'field-content', 'field-button_text', 'field-button_url'],
    custom: ['field-title', 'field-extra_data'],
};

const ALL_FIELDS = [
    'field-title', 'field-subtitle', 'field-content',
    'field-image', 'field-image_alt',
    'field-button_text', 'field-button_url', 'field-extra_data',
];

function toggleFields(inlineRow, selectedType) {
    const visibleFields = SECTION_FIELDS[selectedType] || ALL_FIELDS;

    ALL_FIELDS.forEach(function(fieldClass) {
        const field = inlineRow.querySelector('.' + fieldClass);
        if (!field) return;
        field.style.display = visibleFields.includes(fieldClass) ? '' : 'none';
    });

    inlineRow.querySelectorAll('.inline-group').forEach(function(carouselBlock) {
        carouselBlock.style.display = selectedType === 'gallery' ? '' : 'none';
    });
}

function initRow(row) {
    const select = row.querySelector('select[name$="-section_type"]');
    if (!select) return;
    toggleFields(row, select.value);
    select.addEventListener('change', function() {
        toggleFields(row, this.value);
    });
}

document.addEventListener('DOMContentLoaded', function() {
    document.querySelectorAll('.inline-related:not(.empty-form)').forEach(initRow);

    document.addEventListener('formset:added', function(event) {
        initRow(event.target);
    });
});

import { NextResponse } from 'next/server';
import type { NextRequest } from 'next/server';
import {
  buildRedirectLookup,
  resolveRedirect,
  toRedirectUrl,
  type RedirectRule,
} from '@/lib/redirects';

const API_URL =
  process.env.API_URL ||
  process.env.NEXT_PUBLIC_API_URL ||
  'http://127.0.0.1:8000';

let cachedLookup: ReturnType<typeof buildRedirectLookup> | null = null;
let cacheExpiresAt = 0;
const CACHE_TTL_MS = 60_000;

async function getRedirectLookup() {
  const now = Date.now();
  if (cachedLookup && now < cacheExpiresAt) {
    return cachedLookup;
  }

  try {
    const res = await fetch(`${API_URL}/api/v1/redirects/`, {
      next: { revalidate: 60 },
    });
    if (!res.ok) return cachedLookup ?? new Map();
    const rules = (await res.json()) as RedirectRule[];
    cachedLookup = buildRedirectLookup(rules);
    cacheExpiresAt = now + CACHE_TTL_MS;
    return cachedLookup;
  } catch {
    return cachedLookup ?? new Map();
  }
}

export async function proxy(request: NextRequest) {
  const { pathname } = request.nextUrl;
  const lookup = await getRedirectLookup();
  const hit = resolveRedirect(pathname, lookup);

  if (!hit) {
    return NextResponse.next();
  }

  const destination = toRedirectUrl(hit.to_url, request.nextUrl.origin);
  return NextResponse.redirect(destination, hit.status_code);
}

  ],

import { PageRenderer } from '@/components/PageRenderer';

interface PageData {
  key: string;
  title: string;
  meta_title: string;
  meta_description: string;
  og_image: string | null;
  robots: string;
  sections: any[];
}

const RESERVED_KEYS = ['reports', 'events', 'contacts', 'pages', 'api', 'admin', 'auth', 'search'];

async function getPage(key: string): Promise<PageData | null> {
  try {
    const res = await fetch(`${process.env.API_URL}/api/v1/pages/${key}/`, {
      next: { revalidate: 300 },
    });
    if (!res.ok) return null;
    return res.json();
  } catch {
    return null;
  }
}

export async function generateMetadata({ params }: { params: Promise<{ key: string }> }): Promise<Metadata> {
  const { key } = await params;
  if (RESERVED_KEYS.includes(key)) return { title: 'Сторінку не знайдено' };
  const page = await getPage(key);
  if (!page) return { title: 'Сторінку не знайдено' };
  return {
    title: page.meta_title || page.title,
    description: page.meta_description,
    robots: page.robots,
    openGraph: {
      title: page.meta_title || page.title,
      description: page.meta_description,
      images: page.og_image ? [page.og_image] : [],
    },
  };
}

export default async function DynamicPage({ params }: { params: Promise<{ key: string }> }) {
  const { key } = await params;

  if (RESERVED_KEYS.includes(key)) notFound();

  const page = await getPage(key);
  if (!page) notFound();

  return (
    <main>
      {page.sections?.length > 0 ? (
        <PageRenderer sections={page.sections} />
      ) : (
        <section className="section">
          <div className="container">
            <h1 className="section-title">{page.title}</h1>
            <p style={{ color: 'var(--charcoal)', fontSize: '1.125rem' }}>
              Сторінка поки не має контенту.
            </p>
          </div>
        </section>
      )}
    </main>
  );
}

import { Metadata } from 'next';
import Link from 'next/link';

import { Search, Calendar, FileText, Layout, ArrowRight } from 'lucide-react';

export const metadata: Metadata = {
  title: 'Пошук',
  description: 'Пошук по сайту організації Місто Сили.',
  robots: 'noindex',
};

interface SearchResult {
  type: 'report' | 'event' | 'page';
  id: number;
  title: string;
  slug?: string;
  key?: string;
  preview?: string;
  description?: string;
  meta_description?: string;
  date_published?: string;
  date?: string;
  image?: string | null;
}

interface SearchResponse {
  query: string;
  results: SearchResult[];
  total: number;
  error?: string;
}

async function search(query: string, type?: string): Promise<SearchResponse> {
  if (!query || query.length < 2) {
    return { query, results: [], total: 0 };
  }
  try {
    const params = new URLSearchParams({ q: query });
    if (type) params.set('type', type);
    const res = await fetch(
      `${process.env.API_URL}/api/v1/search/?${params}`,
      { cache: 'no-store' }
    );
    if (!res.ok) return { query, results: [], total: 0 };
    return res.json();
  } catch {
    return { query, results: [], total: 0 };
  }
}

function formatDate(dateStr: string): string {
  return new Date(dateStr).toLocaleDateString('uk-UA', {
    day: 'numeric', month: 'long', year: 'numeric',
  });
}

function getResultHref(result: SearchResult): string {
  if (result.type === 'report') return `/reports/${result.slug}`;
  if (result.type === 'event')  return `/events/${result.slug}`;
  if (result.type === 'page')   return `/${result.key}`;
  return '/';
}

function getResultDate(result: SearchResult): string | null {
  if (result.date_published) return formatDate(result.date_published);
  if (result.date)           return formatDate(result.date);
  return null;
}

function getResultPreview(result: SearchResult): string {
  return result.preview || result.description || result.meta_description || '';
}

function TypeBadge({ type }: { type: string }) {

    report: { label: 'Звіт',     color: 'var(--teal)',     Icon: FileText },
    event:  { label: 'Подія',    color: 'var(--red)',      Icon: Calendar },
    page:   { label: 'Сторінка', color: 'var(--charcoal)', Icon: Layout },
  };
  const c = config[type] || config.page;

  return (
    <span style={{
      display: 'inline-flex', alignItems: 'center', gap: '0.3rem',
      padding: '0.2rem 0.625rem',
      background: c.color, color: '#fff',
      fontSize: '0.75rem', fontWeight: 700,
      textTransform: 'uppercase', letterSpacing: '0.05em',
      flexShrink: 0,
    }}>
      <c.Icon size={11} />
      {c.label}
    </span>
  );
}

export default async function SearchPage({
  searchParams,
}: {
  searchParams: Promise<{ q?: string; type?: string }>;

  const { q = '', type } = await searchParams;
  const data = await search(q, type);

'use client';

import { useEffect, useRef, useState } from 'react';
import { Lightbox } from './Lightbox';

interface ContentWithLightboxProps {
  html: string;
  className?: string;
  style?: React.CSSProperties;
}

export function ContentWithLightbox({ html, className, style }: ContentWithLightboxProps) {
  const containerRef = useRef<HTMLDivElement>(null);
  const [lightboxImages, setLightboxImages] = useState<{ src: string; alt: string }[]>([]);
  const [lightboxIndex, setLightboxIndex] = useState<number | null>(null);

  useEffect(() => {
    const container = containerRef.current;
    if (!container) return;

    const styleImgs = () => {
      const imgs = Array.from(container.querySelectorAll('img'));
      imgs.forEach(img => { img.style.cursor = 'zoom-in'; });
      setLightboxImages(imgs.map(img => ({ src: img.src, alt: img.alt || '' })));
    };
    styleImgs();

    const handleClick = (e: MouseEvent) => {
      const target = e.target as HTMLElement;
      if (target.tagName === 'IMG') {
        const imgs = Array.from(container.querySelectorAll('img'));
        const index = imgs.indexOf(target as HTMLImageElement);
        if (index !== -1) {
          setLightboxImages(imgs.map(img => ({ src: img.src, alt: img.alt || '' })));
          setLightboxIndex(index);
        }
      }
    };

    container.addEventListener('click', handleClick);
    return () => container.removeEventListener('click', handleClick);
  }, [html]);

  return (
    <>
      <div
        ref={containerRef}
        className={className}
        style={style}
        dangerouslySetInnerHTML={{ __html: html }}
      />

      {lightboxIndex !== null && lightboxImages.length > 0 && (
        <Lightbox
          images={lightboxImages}
          initialIndex={lightboxIndex}
          onClose={() => setLightboxIndex(null)}
        />
      )}
    </>
  );
}

'use client';

import { useState, useEffect } from 'react';
import Image from 'next/image';
import { X, ChevronLeft, ChevronRight } from 'lucide-react';

interface LightboxImage {
  src: string;
  alt?: string;
  caption?: string;
}

interface LightboxProps {
  images: LightboxImage[];
  initialIndex?: number;
  onClose: () => void;
}

export function Lightbox({ images, initialIndex = 0, onClose }: LightboxProps) {
  const [current, setCurrent] = useState(initialIndex);

  const prev = () => setCurrent(i => (i > 0 ? i - 1 : images.length - 1));
  const next = () => setCurrent(i => (i < images.length - 1 ? i + 1 : 0));

  useEffect(() => {
    const handler = (e: KeyboardEvent) => {
      if (e.key === 'Escape') onClose();
      if (e.key === 'ArrowLeft') prev();
      if (e.key === 'ArrowRight') next();
    };
    document.addEventListener('keydown', handler);
    document.body.style.overflow = 'hidden';
    return () => {
      document.removeEventListener('keydown', handler);
      document.body.style.overflow = '';
    };
  }, []);

  return (
    <div
      onClick={onClose}
      style={{
        position: 'fixed', inset: 0, zIndex: 9999,
        background: 'rgba(0,0,0,0.92)',
        display: 'flex', alignItems: 'center', justifyContent: 'center',
      }}
    >
      <button
        onClick={onClose}
        aria-label="Закрити"
        style={{
          position: 'absolute', top: '1rem', right: '1rem',
          background: 'rgba(255,255,255,0.1)', border: 'none',
          color: '#fff', cursor: 'pointer', padding: '0.5rem',
          display: 'flex', alignItems: 'center', borderRadius: '2px',
          transition: 'background 0.2s',
        }}
        onMouseEnter={e => (e.currentTarget.style.background = 'rgba(255,255,255,0.2)')}
        onMouseLeave={e => (e.currentTarget.style.background = 'rgba(255,255,255,0.1)')}
      >
        <X size={28} />
      </button>

      {images.length > 1 && (
        <button
          onClick={e => { e.stopPropagation(); prev(); }}
          aria-label="Попереднє"
          style={{
            position: 'absolute', left: '1rem', top: '50%', transform: 'translateY(-50%)',
            background: 'rgba(255,255,255,0.1)', border: 'none',
            color: '#fff', cursor: 'pointer', padding: '0.75rem',
            display: 'flex', alignItems: 'center', borderRadius: '2px',
          }}
        >
          <ChevronLeft size={32} />
        </button>
      )}

      <div
        onClick={e => e.stopPropagation()}
        style={{
          position: 'relative',
          maxWidth: '90vw', maxHeight: '85vh',
          width: '100%', height: '100%',
          display: 'flex', alignItems: 'center', justifyContent: 'center',

'var(--teal)' : 'var(--charcoal)', padding: '0.875rem 0', borderBottom: '1px solid var(--light-gray)', textDecoration: 'none', textTransform: 'uppercase', letterSpacing: '0.05em' }}>
                {link.label}
              </Link>
            );
          })}

          {extraPages.length > 0 && (
            <>
              <div style={{ padding: '0.875rem 0 0.25rem', fontSize: '0.8rem', fontWeight: 700, color: 'var(--teal)', textTransform: 'uppercase', letterSpacing: '0.1em' }}>
                Інше
              </div>
              {extraPages.map(page => (
                <Link key={page.key} href={`/${page.key}`} style={{ display: 'block', fontFamily: 'var(--font-main)', fontSize: '1.125rem', fontWeight: pathname === `/${page.key}` ? 700 : 400, color: pathname === `/${page.key}` ? 'var(--teal)' : 'var(--charcoal)', padding: '0.75rem 0 0.75rem 1rem', borderBottom: '1px solid var(--light-gray)', textDecoration: 'none' }}>
                  {page.title}
                </Link>
              ))}
            </>
          )}
        </nav>
      )}

      <style>{`
        @media (max-width: 768px) {
          .desktop-nav { display: none !important; }
          .mobile-btns { display: flex !important; align-items: center; gap: 0.25rem; }
        }
      `}</style>
    </header>
  );
}

const API_URL =
  process.env.API_URL ||
  process.env.NEXT_PUBLIC_API_URL ||
  'http://127.0.0.1:8000';

export function apiUrl(path: string): string {
  const normalized = path.startsWith('/') ? path : `/${path}`;
  return `${API_URL}${normalized}`;
}

/** Події, звіти — без кешу, щоб зміни з адмінки були одразу на сайті. */
export function fetchLive(path: string, init?: RequestInit) {
  return fetch(apiUrl(path), { ...init, cache: 'no-store' });
}

/** Сторінки, контакти — ISR з revalidate. */
export function fetchCached(path: string, revalidate = 300, init?: RequestInit) {
  return fetch(apiUrl(path), { ...init, next: { revalidate } });
}

export interface RedirectRule {
  from_path: string;
  to_url: string;
  status_code: number;
}

export type RedirectLookup = Map<string, { to_url: string; status_code: number }>;

function pathVariants(path: string): string[] {
  const normalized = path.startsWith('/') ? path : `/${path}`;
  const variants = new Set([normalized]);
  if (normalized.length > 1 && normalized.endsWith('/')) {
    variants.add(normalized.slice(0, -1));
  } else {
    variants.add(`${normalized}/`);
  }
  return [...variants];
}

export function buildRedirectLookup(rules: RedirectRule[]): RedirectLookup {
  const lookup: RedirectLookup = new Map();
  for (const rule of rules) {
    const target = { to_url: rule.to_url, status_code: rule.status_code };
    for (const variant of pathVariants(rule.from_path)) {
      lookup.set(variant, target);
    }
  }
  return lookup;
}

export function resolveRedirect(
  pathname: string,
  lookup: RedirectLookup,
): { to_url: string; status_code: number } | null {
  for (const candidate of pathVariants(pathname)) {
    const hit = lookup.get(candidate);
    if (hit) return hit;
  }
  return null;
}

export function toRedirectUrl(toUrl: string, origin: string): string {
  if (/^https?:\/\//i.test(toUrl)) return toUrl;
  return new URL(toUrl.startsWith('/') ? toUrl : `/${toUrl}`, origin).toString();
}
